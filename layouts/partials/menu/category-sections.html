{{- /* Menu category section pages for home/header. Context: current page. Flat mode: top-level sections with RegularPages. Multi mode on location home: .Sections of that location. Multi mode on site home: empty (chooser only). */ -}} {{- $out := slice -}} {{- $multi := partial "menu/multi-location-enabled.html" . -}} {{- if $multi -}} {{- $locSlug := partial "menu/location-slug.html" . -}} {{- if $locSlug -}} {{- $locPage := site.GetPage (printf "/%s" $locSlug) -}} {{- with $locPage -}} {{- range .Sections -}} {{- $titleurl := .RelPermalink -}} {{- if and .RegularPages (not (partial "menu/is-excluded-section.html" $titleurl)) -}} {{- $out = $out | append . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- else -}} {{- /* Top-level catalog sections. Use site.Sections — Home.Sections can drop mounted .generated folders after a live merge. Skip /{location}/ trees. */ -}} {{- $locSlugs := slice -}} {{- with site.Data.locations -}} {{- if reflect.IsSlice . -}} {{- range . -}}{{- with .slug -}}{{- $locSlugs = $locSlugs | append . -}}{{- end -}}{{- end -}} {{- else if .locations -}} {{- range .locations -}}{{- with .slug -}}{{- $locSlugs = $locSlugs | append . -}}{{- end -}}{{- end -}} {{- end -}} {{- end -}} {{- range site.Sections.ByWeight -}} {{- if .RegularPages -}} {{- $titleurl := .RelPermalink -}} {{- $first := index (split (strings.TrimPrefix "/" $titleurl) "/") 0 -}} {{- if and (not (in $locSlugs $first)) (not (partial "menu/is-excluded-section.html" $titleurl)) -}} {{- $out = $out | append . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $out -}}
