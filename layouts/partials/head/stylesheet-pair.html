{{- /* Load mobile-first base CSS, then companion *.lg.css for larger screens. Params: href - base stylesheet path, e.g. "/css/header.css" sync - if true, blocking ; else async preload lg - if false, skip companion (default true). Ignored for non-/css/*.css */ -}} {{- $href := .href -}} {{- $sync := .sync | default false -}} {{- $withLg := true -}} {{- if isset . "lg" -}}{{- $withLg = .lg -}}{{- end -}} {{- if $sync -}} {{- else -}} {{ partial "head/async-stylesheet.html" (dict "href" $href) }} {{- end -}} {{- if $withLg -}} {{- $lgHref := "" -}} {{- if hasSuffix $href ".css" -}} {{- $lgHref = printf "%s.lg.css" (strings.TrimSuffix ".css" $href) -}} {{- end -}} {{- if $lgHref -}} {{- if $sync -}} {{- else -}} {{ partial "head/async-stylesheet.html" (dict "href" $lgHref) }} {{- end -}} {{- end -}} {{- end -}}
