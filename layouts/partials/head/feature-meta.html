{{- /* Home title and description from the client name and enabled features. */ -}} {{- $siteName := partial "site-name.html" . | strings.TrimSpace -}} {{- $f := site.Params.features | default dict -}} {{- $os := site.Params.orderingsystem | default dict -}} {{- $pay := site.Params.paymentService | default dict -}} {{- $del := site.Params.deliveryService | default dict -}} {{- $ordering := default false $os.enabled -}} {{- $payments := and (default false $f.payments) (default false $pay.enabled) -}} {{- $delivery := and (default false $f.delivery) (default false $del.enabled) -}} {{- $parts := slice "Digital Menu" -}} {{- if $ordering -}} {{- $parts = $parts | append "Online Ordering" -}} {{- end -}} {{- if $delivery -}} {{- $parts = $parts | append "Delivery" -}} {{- end -}} {{- if $payments -}} {{- $parts = $parts | append "Payments" -}} {{- end -}} {{- $n := len $parts -}} {{- $phrase := index $parts 0 -}} {{- if eq $n 2 -}} {{- $phrase = printf "%s & %s" (index $parts 0) (index $parts 1) -}} {{- else if gt $n 2 -}} {{- $last := index $parts (sub $n 1) -}} {{- $head := first (sub $n 1) $parts -}} {{- $phrase = printf "%s & %s" (delimit $head ", ") $last -}} {{- end -}} {{- return (dict "phrase" $phrase "description" (printf "%s. %s." $siteName $phrase)) -}}
